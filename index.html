import numpy as np

def retrieve_grounded_context(query_vec: np.ndarray, doc_embeddings: np.ndarray, chunks: list[str], top_k: int = 3):
    """Calculates cosine similarity ranking locally with zero cloud transmission."""
    norm_query = query_vec / np.linalg.norm(query_vec)
    norm_docs = doc_embeddings / np.linalg.norm(doc_embeddings, axis=1, keepdims=True)
    similarity_scores = np.dot(norm_docs, norm_query)
    
    top_indices = np.argsort(similarity_scores)[::-1][:top_k]
    return [{"chunk": chunks[i], "score": float(similarity_scores[i])} for i in top_indices]

class MilestoneStateMachine {
    private const ALLOWED_TRANSITIONS = [
        'draft'     => ['submitted'],
        'submitted' => ['under_review', 'revision_requested'],
        'under_review' => ['approved', 'rejected'],
        'approved'  => ['paid']
    ];

    public function transition(Milestone $milestone, string $targetState, User $actor): bool {
        if (!in_array($targetState, self::ALLOWED_TRANSITIONS[$milestone->status] ?? [])) {
            throw new InvalidStateTransitionException("Cannot move from {$milestone->status} to {$targetState}");
        }
        
        DB::transaction(function() use ($milestone, $targetState, $actor) {
            $milestone->update(['status' => $targetState]);
            AuditLog::create([
                'entity_id' => $milestone->id,
                'from_state' => $milestone->getOriginal('status'),
                'to_state' => $targetState,
                'actor_id' => $actor->id
            ]);
        });
        return true;
    }
}

import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

class HybridRecommender:
    def __init__(self, user_item_matrix, item_features):
        self.matrix = user_item_matrix
        self.item_sim = cosine_similarity(item_features)

    def predict_top_k(self, user_vector: np.ndarray, k: int = 5):
        # Weighted dot-product across interaction history and item feature space
        scores = user_vector.dot(self.item_sim) / (np.linalg.norm(user_vector) + 1e-9)
        ranked_indices = np.argsort(scores)[::-1][:k]
        return ranked_indices, scores[ranked_indices]

export interface TaskPayload {
  title: string;
  priority: 'low' | 'medium' | 'high' | 'critical';
  deadline: string;
}

export function validateTaskPayload(payload: Partial): TaskPayload {
  if (!payload.title || payload.title.trim().length < 3) {
    throw new ValidationError('Task title must be at least 3 characters.');
  }
  const validPriorities = ['low', 'medium', 'high', 'critical'];
  if (!payload.priority || !validPriorities.includes(payload.priority)) {
    throw new ValidationError('Invalid priority tier.');
  }
  return payload as TaskPayload;
}
